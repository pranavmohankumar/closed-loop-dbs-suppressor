4. Core Technical Starter Skeleton

import numpy as np
from scipy.signal import butter, lfilter

def real_time_beta_power(lfp_sample_window, fs=1000):
    b, a = butter(2, [13/(fs/2), 30/(fs/2)], btype='band')
    beta_filtered = lfilter(b, a, lfp_sample_window)
    beta_power = np.mean(beta_filtered**2)
    return beta_power
